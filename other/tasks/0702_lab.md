## 2. ТЕОРЕТИЧЕСКАЯ ЧАСТЬ

### 2.1. Что такое Retrofit?

**Retrofit** — типобезопасный HTTP-клиент для Android и Java, разработанный компанией Square. Он позволяет преобразовывать HTTP-API в интерфейсы Kotlin/Java.

**Преимущества Retrofit:**
- Простота использования (описание API через интерфейсы).
- Встроенная поддержка корутин (`suspend`-функции).
- Гибкость (поддержка различных конвертеров: Gson, Moshi, kotlinx.serialization).
- Интеграция с OkHttp для настройки сетевых запросов.

package com.example.todolist.data.remote

import okhttp3.OkHttpClient
import okhttp3.logging.HttpLoggingInterceptor
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory
import java.util.concurrent.TimeUnit

/**
 * Клиент Retrofit для выполнения сетевых запросов.
 *
 * @author [ВАШЕ ИМЯ]
 * @since 2026-09-03
 */
object RetrofitClient {

    private const val BASE_URL = "https://jsonplaceholder.typicode.com/"

    private val client: OkHttpClient by lazy {
        OkHttpClient.Builder()
            .addInterceptor(
                HttpLoggingInterceptor().apply {
                    level = HttpLoggingInterceptor.Level.BODY
                }
            )
            .connectTimeout(30, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .build()
    }

    val apiService: ApiService by lazy {
        Retrofit.Builder()
            .baseUrl(BASE_URL)
            .client(client)
            .addConverterFactory(GsonConverterFactory.create())
            .build()
            .create(ApiService::class.java)
    }
}

package com.example.todolist.data.repository

import com.example.todolist.data.local.Task
import com.example.todolist.data.local.TaskDao
import com.example.todolist.data.remote.ApiService
import com.example.todolist.data.remote.TaskDTO
import kotlinx.coroutines.flow.Flow
import java.io.IOException

/**
 * Репозиторий для работы с задачами.
 * Объединяет локальные (Room) и удалённые (API) источники данных.
 *
 * @param taskDao DAO для работы с локальной БД.
 * @param apiService Сервис для работы с удалённым API.
 *
 * @author [ВАШЕ ИМЯ]
 * @since 2026-09-03
 */
class TaskRepository(
    private val taskDao: TaskDao,
    private val apiService: ApiService
) {

    /**
     * Получить все задачи (из локального кеша).
     * Подписывается на Flow из Room.
     */
    fun getAllTasks(): Flow<List<Task>> = taskDao.getAllTasks()

    /**
     * Загрузить задачи из сети и сохранить в кеш.
     *
     * @return Список задач или null при ошибке.
     */
    suspend fun fetchTasksFromNetwork(): List<Task>? {
        return try {
            // 1. Загрузить данные из API
            val taskDTOs = apiService.getTasks()
            
            // 2. Преобразовать DTO в Entity
            val tasks = taskDTOs.map { dto ->
                Task(
                    id = dto.id,
                    title = dto.title,
                    isCompleted = dto.completed,
                    userId = dto.userId,
                    isSynced = true
                )
            }
            
            // 3. Сохранить в базу данных
            taskDao.insertAll(tasks)
            
            tasks
        } catch (e: IOException) {
            // Ошибка сети
            null
        } catch (e: Exception) {
            // Другие ошибки
            null
        }
    }

    /**
     * Добавить новую задачу (локально).
     */
    suspend fun insertTask(task: Task) {
        taskDao.insert(task)
    }

    /**
     * Обновить задачу.
     */
    suspend fun updateTask(task: Task) {
        taskDao.update(task)
    }

    /**
     * Удалить задачу.
     */
    suspend fun deleteTask(task: Task) {
        taskDao.delete(task)
    }

    /**
     * Переключить статус выполнения задачи.
     */
    suspend fun toggleTaskCompletion(task: Task) {
        val updatedTask = task.copy(isCompleted = !task.isCompleted)
        updateTask(updatedTask)
    }
}

class TaskViewModel(
    private val repository: TaskRepository
) : ViewModel() {

    private val _tasks = MutableStateFlow<List<Task>>(emptyList())
    val tasks: StateFlow<List<Task>> = _tasks.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()

    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()

    private val _isRefreshing = MutableStateFlow(false)
    val isRefreshing: StateFlow<Boolean> = _isRefreshing.asStateFlow()

    init {
        // Подписываемся на изменения в БД
        viewModelScope.launch {
            repository.getAllTasks().collect { taskList ->
                _tasks.value = taskList
            }
        }
        
        // Загружаем начальные данные
        loadTasks()
    }

    /**
     * Загрузить задачи.
     * Сначала показывает кеш, затем обновляет из сети.
     */
    fun loadTasks() {
        viewModelScope.launch {
            _isLoading.value = true
            _errorMessage.value = null
            
            try {
                // 1. Сначала пробуем загрузить из кеша (уже подписаны через Flow)
                // 2. Затем обновляем из сети
                refreshFromNetwork()
            } catch (e: Exception) {
                _errorMessage.value = "Ошибка загрузки данных: ${e.message}"
            } finally {
                _isLoading.value = false
            }
        }
    }

     * Обновить данные из сети.

                            Text("Загрузить из сети")
                        }
                    }
                }
                else -> {
                    LazyColumn(
                        modifier = Modifier.fillMaxSize()
                    ) {
                        items(
                            items = tasks,
                            key = { it.id }
                        ) { task ->
                            TaskItem(
                                task = task,
                                onToggle = {
                                    viewModel.toggleTaskCompletion(task)
                                }
                            )
                        }
                    }
                }
            }
        }

        // Диалог добавления задачи
        if (showDialog) {
            AlertDialog(
                onDismissRequest = { showDialog = false },
                title = { Text("Новая задача") },
                text = {
                    OutlinedTextField(
                        value = newTaskTitle,
                        onValueChange = { newTaskTitle = it },
                        label = { Text("Название задачи") },
                        singleLine = true
                    )
                },
                confirmButton = {
                    TextButton(
                        onClick = {
                            if (newTaskTitle.isNotBlank()) {
                                viewModel.addTask(newTaskTitle)
                                newTaskTitle = ""
                                showDialog = false
                            }
                        }
                    ) {
                        Text("Добавить")
                    }
                },
                dismissButton = {
                    TextButton(onClick = { showDialog = false }) {
                        Text("Отмена")
                    }
                }
            )
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun TaskListScreen(viewModel: TaskViewModel) {
    val errorMessage by viewModel.errorMessage.collectAsState()
    val snackbarHostState = remember { SnackbarHostState() }
    
    if (errorMessage != null) {
        LaunchedEffect(errorMessage) {
            snackbarHostState.showSnackbar(errorMessage!!)
            viewModel.clearError()
        }
    }
    
    Scaffold(
        snackbarHost = { SnackbarHost(snackbarHostState) }
    ) {
        // ...
    }
}

@Composable
fun TaskListScreen(viewModel: TaskViewModel) {
    val isRefreshing by viewModel.isRefreshing.collectAsState()
    
    Box(
        modifier = Modifier
            .fillMaxSize()
            .pullRefresh(
                isRefreshing = isRefreshing,
                onRefresh = { viewModel.refresh() }
            )
    ) {
        // ...
        PullRefreshIndicator(
            isRefreshing = isRefreshing,
            state = rememberPullRefreshState(isRefreshing, { viewModel.refresh() })
        )
    }
}
